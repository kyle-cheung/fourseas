# FX Rates Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fetch required Frankfurter rates after transaction sync and calculate USD values in `v_transactions` with an ASOF lookup.

**Architecture:** Keep HTTP, persistence, and orchestration separate. A Frankfurter client returns exact decimal rate models, the store owns `fx_rates` and currency bounds, and command orchestration selects one range for each required currency. Raw transactions keep provider values; the view calculates conversion fields from the latest rate on or before each transaction date.

**Tech Stack:** Go 1.25, DuckDB 1.5.5 through `duckdb-go/v2`, `shopspring/decimal`, `net/http`, Frankfurter v2 JSON API.

```go
package store

import (
	"context"
	"database/sql"
	"fmt"
	"strings"

	"github.com/kyle-cheung/fourseas/providence/internal/model"
)

func normalizeCurrency(currency string) string {
	return strings.ToUpper(strings.TrimSpace(currency))
}

func (s *Store) RequiredFXCurrencies(ctx context.Context, baseCurrency string) ([]model.FXCurrency, error) {
	rows, err := s.db.QueryContext(ctx, `
		SELECT upper(currency), min(date)
		FROM transactions
		WHERE currency IS NOT NULL AND trim(currency) <> '' AND upper(currency) <> upper(?)
		GROUP BY upper(currency)
		ORDER BY upper(currency)`, normalizeCurrency(baseCurrency))
	if err != nil {
		return nil, fmt.Errorf("query currencies that need FX rates: %w", err)
	}
	defer rows.Close()
	var out []model.FXCurrency
	for rows.Next() {
		var currency model.FXCurrency
		if err := rows.Scan(&currency.Currency, &currency.OldestTransactionDate); err != nil {
			return nil, fmt.Errorf("scan currency that needs FX rates: %w", err)
		}
		out = append(out, currency)
	}
	return out, rows.Err()
}

func (s *Store) FXRateBounds(ctx context.Context, currency, baseCurrency string) (model.FXRateBounds, error) {
	var oldest, newest sql.NullTime
	err := s.db.QueryRowContext(ctx, `
		SELECT min(date), max(date) FROM fx_rates
		WHERE currency = ? AND base_currency = ?`,
		normalizeCurrency(currency), normalizeCurrency(baseCurrency)).Scan(&oldest, &newest)
	if err != nil {
		return model.FXRateBounds{}, fmt.Errorf("query FX rate bounds for %s/%s: %w", currency, baseCurrency, err)
	}
	return model.FXRateBounds{OldestDate: timePtr(oldest), NewestDate: timePtr(newest)}, nil
}

func (s *Store) UpsertFXRates(ctx context.Context, rates []model.FXRate) error {
	return s.inTx(ctx, func(dbtx execer) error {
		if len(rates) == 0 {
			return nil
		}
		stmt, err := dbtx.PrepareContext(ctx, `
			INSERT INTO fx_rates (date, currency, base_currency, rate)
			VALUES (?, ?, ?, ?)
			ON CONFLICT (date, currency, base_currency) DO UPDATE SET rate = excluded.rate`)
		if err != nil {
			return fmt.Errorf("prepare FX rate upsert: %w", err)
		}
		defer stmt.Close()
		for _, rate := range rates {
			_, err := stmt.ExecContext(ctx, rate.Date, normalizeCurrency(rate.Currency),
				normalizeCurrency(rate.BaseCurrency), decimalArg(rate.Rate))
			if err != nil {
				return fmt.Errorf("upsert FX rate %s/%s on %s: %w", rate.Currency,
					rate.BaseCurrency, rate.Date.Format("2006-01-02"), err)
			}
		}
		return nil
	})
}
```

```go
package frankfurter

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/url"
	"strings"
	"time"

	"github.com/kyle-cheung/fourseas/providence/internal/model"
	"github.com/shopspring/decimal"
)

const DefaultBaseURL = "https://api.frankfurter.dev"

type Client struct {
	httpClient *http.Client
	baseURL    string
}

func NewClient(httpClient *http.Client, baseURL string) *Client {
	if httpClient == nil {
		httpClient = http.DefaultClient
	}
	return &Client{httpClient: httpClient, baseURL: strings.TrimRight(baseURL, "/")}
}

func validCurrency(currency string) bool {
	if len(currency) != 3 {
		return false
	}
	for _, char := range currency {
		if char < 'A' || char > 'Z' {
			return false
		}
	}
	return true
}

func (c *Client) Rates(ctx context.Context, currency, baseCurrency string, from, to time.Time) ([]model.FXRate, error) {
	currency = strings.ToUpper(strings.TrimSpace(currency))
	baseCurrency = strings.ToUpper(strings.TrimSpace(baseCurrency))
	if !validCurrency(currency) || !validCurrency(baseCurrency) {
		return nil, fmt.Errorf("currency codes must contain three ASCII letters")
	}
	if from.After(to) {
		return nil, fmt.Errorf("FX range starts after it ends")
	}
	query := url.Values{
		"base": {currency}, "quotes": {baseCurrency},
		"from": {from.Format("2006-01-02")}, "to": {to.Format("2006-01-02")},
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.baseURL+"/v2/rates?"+query.Encode(), nil)
	if err != nil {
		return nil, fmt.Errorf("build Frankfurter request: %w", err)
	}
	resp, err := c.httpClient.Do(req)
	if err != nil {
		return nil, fmt.Errorf("fetch %s/%s rates: %w", currency, baseCurrency, err)
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return nil, fmt.Errorf("fetch %s/%s rates: HTTP %s", currency, baseCurrency, resp.Status)
	}
	var payload []struct {
		Date  string      `json:"date"`
		Base  string      `json:"base"`
		Quote string      `json:"quote"`
		Rate  json.Number `json:"rate"`
	}
	decoder := json.NewDecoder(resp.Body)
	decoder.UseNumber()
	if err := decoder.Decode(&payload); err != nil {
		return nil, fmt.Errorf("decode %s/%s rates: %w", currency, baseCurrency, err)
	}
	rates := make([]model.FXRate, 0, len(payload))
	for _, row := range payload {
		date, err := time.Parse("2006-01-02", row.Date)
		if err != nil || date.Before(from) || date.After(to) {
			return nil, fmt.Errorf("Frankfurter returned invalid date %q", row.Date)
		}
		rowBase := strings.ToUpper(strings.TrimSpace(row.Base))
		rowQuote := strings.ToUpper(strings.TrimSpace(row.Quote))
		if rowBase != currency || rowQuote != baseCurrency {
			return nil, fmt.Errorf("Frankfurter returned %s/%s, want %s/%s", rowBase, rowQuote, currency, baseCurrency)
		}
		rate, err := decimal.NewFromString(row.Rate.String())
		if err != nil || !rate.IsPositive() {
			return nil, fmt.Errorf("Frankfurter returned invalid rate %q", row.Rate)
		}
		rates = append(rates, model.FXRate{Date: date, Currency: rowBase, BaseCurrency: rowQuote, Rate: rate})
	}
	return rates, nil
}
```

```go
package main

import (
	"context"
	"fmt"
	"time"

	"github.com/kyle-cheung/fourseas/providence/internal/model"
)

type fxStore interface {
	RequiredFXCurrencies(context.Context, string) ([]model.FXCurrency, error)
	FXRateBounds(context.Context, string, string) (model.FXRateBounds, error)
	UpsertFXRates(context.Context, []model.FXRate) error
}

type fxSource interface {
	Rates(context.Context, string, string, time.Time, time.Time) ([]model.FXRate, error)
}

type fxResult struct {
	Currency string
	Rows     int
	Err      error
}

func fxStartDate(oldestTransaction time.Time, bounds model.FXRateBounds, today time.Time) time.Time {
	start := oldestTransaction
	if bounds.OldestDate != nil && bounds.NewestDate != nil && !oldestTransaction.Before(*bounds.OldestDate) {
		start = *bounds.NewestDate
	}
	if start.After(today) {
		return today
	}

func syncFX(ctx context.Context, db fxStore, source fxSource, baseCurrency string, now time.Time) ([]fxResult, error) {
	today := time.Date(now.UTC().Year(), now.UTC().Month(), now.UTC().Day(), 0, 0, 0, 0, time.UTC)
	required, err := db.RequiredFXCurrencies(ctx, baseCurrency)
	if err != nil {
		return nil, err
	}
	results := make([]fxResult, 0, len(required))
	for _, currency := range required {
		if err := ctx.Err(); err != nil {
			return results, err
		}
		result := fxResult{Currency: currency.Currency}
		bounds, err := db.FXRateBounds(ctx, currency.Currency, baseCurrency)
		if err == nil {
			start := fxStartDate(currency.OldestTransactionDate, bounds, today)
			var rates []model.FXRate
			rates, err = source.Rates(ctx, currency.Currency, baseCurrency, start, today)
			if err == nil {
				err = db.UpsertFXRates(ctx, rates)
				result.Rows = len(rates)
			}
		}
		if err != nil {
			if ctx.Err() != nil {
				return results, ctx.Err()
			}
			result.Err = fmt.Errorf("%s to %s: %w", currency.Currency, baseCurrency, err)
		}
		results = append(results, result)
	}
	return results, nil
}
```

```go
func runFXPhase(ctx context.Context, db fxStore, source fxSource, now time.Time, strict bool, out io.Writer) error {
	results, err := syncFX(ctx, db, source, model.BaseCurrency, now)
	if err != nil {
		if ctx.Err() != nil || strict {
			return err
		}
		fmt.Fprintf(out, "FX warning: %v\n", err)
		return nil
	}
	if len(results) == 0 {
		fmt.Fprintln(out, "FX: no non-USD transaction currencies need rates")
		return nil
	}
	var failures []error
	for _, result := range results {
		if result.Err != nil {
			fmt.Fprintf(out, "FX %s -> %s warning: %v\n", result.Currency, model.BaseCurrency, result.Err)
			failures = append(failures, result.Err)
			continue
		}
		fmt.Fprintf(out, "FX %s -> %s: %d rates\n", result.Currency, model.BaseCurrency, result.Rows)
	}
	if strict && len(failures) > 0 {
		return errors.Join(failures...)
	}
	return nil
}
```
