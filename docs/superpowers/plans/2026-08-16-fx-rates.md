# FX Rates Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fetch Frankfurter rates after transaction sync and calculate USD values in `v_transactions` with an ASOF lookup.

**Architecture:** The store owns normalized `fx_rates`; a small HTTP client owns the Frankfurter contract; command orchestration refreshes each non-USD currency from its oldest transaction date through today. Raw transactions keep provider values, and the view calculates conversion fields from the latest rate on or before each transaction date.

**Tech Stack:** Go 1.25, DuckDB 1.5.5 through `duckdb-go/v2`, `shopspring/decimal`, `net/http`, Frankfurter v2.

```go
func (s *Store) RequiredFXCurrencies(ctx context.Context, baseCurrency string) ([]model.FXCurrency, error) {
	rows, err := s.db.QueryContext(ctx, `
		SELECT currency, min(date)
		FROM transactions
		WHERE currency IS NOT NULL AND currency <> '' AND currency <> ?
		GROUP BY currency ORDER BY currency`, model.NormalizeCurrency(baseCurrency))
	if err != nil {
		return nil, fmt.Errorf("query currencies that need FX rates: %w", err)
	}
	defer rows.Close()
	var result []model.FXCurrency
	for rows.Next() {
		var currency model.FXCurrency
		if err := rows.Scan(&currency.Currency, &currency.OldestTransactionDate); err != nil {
			return nil, fmt.Errorf("scan currency that needs FX rates: %w", err)
		}
		result = append(result, currency)
	}
	return result, rows.Err()
}

func (s *Store) UpsertFXRates(ctx context.Context, rates []model.FXRate) error {
	return s.inTx(ctx, func(dbtx execer) error {
		if len(rates) == 0 {
			return nil
		}
		stmt, err := dbtx.PrepareContext(ctx, `
			INSERT INTO fx_rates (date, currency, base_currency, rate) VALUES (?, ?, ?, ?)
			ON CONFLICT (date, currency, base_currency) DO UPDATE SET rate = excluded.rate`)
		if err != nil {
			return fmt.Errorf("prepare FX rate upsert: %w", err)
		}
		defer stmt.Close()
		for _, rate := range rates {
			_, err := stmt.ExecContext(ctx, rate.Date, model.NormalizeCurrency(rate.Currency),
				model.NormalizeCurrency(rate.BaseCurrency), decimalArg(rate.Rate))
			if err != nil {
				return fmt.Errorf("upsert FX rate on %s: %w", rate.Date.Format("2006-01-02"), err)
			}
		}
		return nil
	})
}
```
